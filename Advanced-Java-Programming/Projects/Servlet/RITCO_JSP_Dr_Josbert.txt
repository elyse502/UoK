<script>
  let remaining=20,booked=0;
  const tickets=[];
  let selected=-1;

  function pad(n){return String(n).padStart(2,'0')}
  function today(){const d=new Date();return d.getFullYear()+'-'+pad(d.getMonth()+1)+'-'+pad(d.getDate())}
  function nowTime(){const d=new Date();return pad(d.getHours())+':'+pad(d.getMinutes())}
  function genCode(){return 'TKT-'+(Math.floor(Math.random()*9000)+1000)}

  document.getElementById('fDate').value=today();
  document.getElementById('fTime').value=nowTime();
  document.getElementById('fCode').value=genCode();

  function updateCards(){
    document.getElementById('cAvail').textContent=remaining;
    document.getElementById('cBooked').textContent=booked;
    document.getElementById('seatsLbl').textContent='Remaining Seats: '+remaining;
  }

  function bookTicket(){
    const name=document.getElementById('fName').value.trim();
    const phone=document.getElementById('fPhone').value.trim();
    const price=document.getElementById('fPrice').value.trim();
    if(!name||!phone||!price){alert('Please fill in Name, Phone and Price.');return}
    if(remaining<=0){alert('All seats are fully booked.');return}
    const t={
      code:document.getElementById('fCode').value,
      name,phone,price,
      status:document.getElementById('fStatus').value,
      route:document.getElementById('fRoute').value,
      date:document.getElementById('fDate').value,
      time:document.getElementById('fTime').value
    };
    tickets.push(t);
    remaining--;booked++;
    addRow(t,tickets.length-1);
    updateCards();
    clearForm();
    alert('Ticket booked successfully!\nCode: '+t.code);
  }

  function addRow(t,idx){
    const tb=document.getElementById('tbody');
    const tr=document.createElement('tr');
    tr.dataset.idx=idx;
    tr.innerHTML=`<td>${t.code}</td><td>${t.name}</td><td>${t.phone}</td><td>${t.price}</td><td>${t.status}</td><td>${t.route}</td><td>${t.date}</td><td>${t.time}</td>`;
    tr.onclick=function(){
      document.querySelectorAll('#tbody tr').forEach(r=>r.classList.remove('selected'));
      tr.classList.add('selected');
      selected=idx;
      showDetails(idx);
    };
    tb.appendChild(tr);
  }

  function clearForm(){
    document.getElementById('fName').value='';
    document.getElementById('fPhone').value='';
    document.getElementById('fPrice').value='';
    document.getElementById('fDate').value=today();
    document.getElementById('fTime').value=nowTime();
    document.getElementById('fStatus').selectedIndex=0;
    document.getElementById('fRoute').selectedIndex=0;
    document.getElementById('fCode').value=genCode();
    selected=-1;
    document.querySelectorAll('#tbody tr').forEach(r=>r.classList.remove('selected'));
  }

  function showDetails(idx){
    const t=tickets[idx];
    const rows=[
      ['Company Name','RITCO'],['Email','ritco@gmail.com'],['Tel','+250 078000000'],
      ['Route',t.route],['Date',t.date],['Time',t.time],
      ['Ticket Code',t.code],['Passenger',t.name],['Phone',t.phone],
      ['Price',t.price+' RWF'],['Status',t.status],['Cashier','Mukamana Aline']
    ];
    document.getElementById('modalBody').innerHTML=rows.map(r=>`<div class="modal-row"><span class="lbl">${r[0]}:</span><span class="val">${r[1]}</span></div>`).join('');
    document.getElementById('modal').classList.add('open');
  }

  function closeModal(){document.getElementById('modal').classList.remove('open')}

  function printTicket(){
    if(selected<0){alert('Please click a ticket row in the table first, then press Print.');return}
    const t=tickets[selected];
    const now=new Date();
    const ts=now.getFullYear()+'-'+pad(now.getMonth()+1)+'-'+pad(now.getDate())+' '+pad(now.getHours())+':'+pad(now.getMinutes())+':'+pad(now.getSeconds());
    const D='================================================';
    const H='################################################';
    const content=[
      H,'           RITCO COMPANY','         BUS TICKET RECEIPT',H,'',
      '  Company     : RITCO','  Email       : ritco@gmail.com',
      '  Tel         : +250 078000000','  Country     : RWANDA','',
      D,'  TICKET DETAILS',D,
      '  Ticket Code : '+t.code,'  Passenger   : '+t.name,
      '  Phone       : '+t.phone,'  Route       : '+t.route,
      '  Date        : '+t.date,'  Time        : '+t.time,
      '  Price       : '+t.price+' RWF','  Status      : '+t.status,
      '  Cashier     : Mukamana Aline',D,'',
      '  Printed on  : '+ts,'',
      '     THANK YOU FOR BOOKING WITH RITCO!',
      '       Have a safe and pleasant journey!',H
    ].join('\r\n');
    const blob=new Blob([content],{type:'text/plain'});
    const url=URL.createObjectURL(blob);
    const a=document.createElement('a');
    a.href=url;a.download='RITCO_'+t.code+'.txt';
    a.click();URL.revokeObjectURL(url);
  }

  document.getElementById('modal').addEventListener('click',function(e){if(e.target===this)closeModal()});
</script>
