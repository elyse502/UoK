            {PX+120,  PY+PH/4},        // 1 DEF

            {PX+240,  PY+PH/2},        // 3 MID

            bluePlayers[i].homeY = bluePos[i][1];
        }

        // Red team (right side)
        int[][] redPos = {
            {PX+PW-30, PY+PH/2},
            {PX+PW-120, PY+PH/4},
            {PX+PW-120, PY+3*PH/4},
            {PX+PW-240, PY+PH/2},

            redPlayers[i].homeX = redPos[i][0];
            redPlayers[i].homeY = redPos[i][1];
        }

    }

    void kickOff(boolean blueKicksOff) {
        ball = new Ball(PX+PW/2, PY+PH/2, 0, 0);
        // Reset positions slightly
        for (Player p : bluePlayers) { p.x=p.homeX; p.y=p.homeY; p.state=PlayerState.IDLE; }
        for (Player p : redPlayers)  { p.x=p.homeX; p.y=p.homeY; p.state=PlayerState.IDLE; }
        controlledIdx = 4;
    }

    // ══════════════════════════════════════════════════════════════════
    //  GAME LOOP
    // ══════════════════════════════════════════════════════════════════
    @Override public void actionPerformed(ActionEvent e) {
        tick++;
        // Crowd always bobs
        for (int i=0;i<120;i++) crowdBob[i] += 0.06f + rng.nextFloat()*0.02f;

        switch(screen){
            case PLAYING    -> updatePlaying();
            case GOAL_ANIM  -> updateGoalAnim();
            case HALFTIME   -> { halfTimeTimer++; if(halfTimeTimer>240){ startSecondHalf(); } }
            default -> {}
        }

        particles.removeIf(p->{ p.update(); return p.dead; });
        floaters.removeIf(f->{ f.update(); return f.dead; });
        if (screenShake>0) screenShake--;
        repaint();
    }

    void updatePlaying() {
        matchTimer++;
        int halfFrames = HALF_DURATION * FPS;

        // Half time check
        if (matchTimer >= halfFrames && half==1) {
            sound.play("whistle");
            screen = Screen.HALFTIME;
            halfTimeTimer = 0;
            return;
        }
        if (matchTimer >= halfFrames && half==2) {
            sound.play("whistle");
            sound.play("whistle");
            // Full time
            saveScore(Math.max(scoreBlue,scoreRed));
            if(scoreBlue>hiScore||scoreRed>hiScore){
                hiScore=Math.max(scoreBlue,scoreRed); saveHiScore();
            }
            screen = Screen.FULL_TIME;
            return;
        }

        // Tab: switch controlled player
        if (kTab) { switchControl(); kTab=false; }

        // Player input
        Player cp = bluePlayers[controlledIdx];
        float dx=0, dy=0;
        if (kLeft)  dx=-PLAYER_SPEED;
        if (kRight) dx= PLAYER_SPEED;
        if (kUp)    dy=-PLAYER_SPEED;
        if (kDown)  dy= PLAYER_SPEED;
        if (dx!=0&&dy!=0){ dx*=0.707f; dy*=0.707f; }

        // Move controlled player
        if (dx!=0||dy!=0) {
            cp.x = clamp(cp.x+dx, PX+PLAYER_R+1, PX+PW-PLAYER_R-1);
            cp.y = clamp(cp.y+dy, PY+PLAYER_R+1, PY+PH-PLAYER_R-1);
            cp.state=PlayerState.RUN;
            cp.facingAngle = (float)Math.atan2(dy,dx);
        } else { cp.state=PlayerState.IDLE; }

        // Shoot / Kick logic
        boolean spaceNow = kSpace;
        if (spaceNow && !kSpacePrev) { shootPower=0f; spaceHeld=true; }
        if (spaceNow && spaceHeld)   shootPower = Math.min(maxPower, shootPower+0.025f);
        if (!spaceNow && kSpacePrev && spaceHeld) {
            // Release — kick!
            tryKick(cp, shootPower);
            shootPower=0f; spaceHeld=false;
        }
        kSpacePrev = spaceNow;

        // AI for other blue players
        float aiSpd = aiSpeed();
        for (int i=0;i<5;i++) {
            if (i==controlledIdx) continue;
            blueAI(bluePlayers[i], i, aiSpd);
        }
        // AI for red team
        for (int i=0;i<5;i++) redAI(redPlayers[i], i, aiSpd);

        // Update ball
        ball.update();

        // Kick near players
        autoKickAI();

        // Collision: players vs ball
        for (Player p : bluePlayers) p.update();
        for (Player p : redPlayers)  p.update();

        // Possession tracking
        Player closest = closestToBall(bluePlayers);
        Player closestR = closestToBall(redPlayers);
        float dB = dist(closest.x,closest.y,ball.x,ball.y);
        float dR = dist(closestR.x,closestR.y,ball.x,ball.y);
        if (dB < dR) possession[0]++; else possession[1]++;

        // Switch control to nearest blue player to ball automatically (if controlled is far)
        if (dist(cp.x,cp.y,ball.x,ball.y) > 140) {
            int nearest = nearestBlueIdx();
            if (nearest != controlledIdx) controlledIdx = nearest;
        }

        // Goal check
        checkGoal();
        // Ball out of bounds
        keepBallInBounds();
    }

    float aiSpeed(){
        return difficulty==Difficulty.EASY ? AI_SPEED_EASY :
               difficulty==Difficulty.MEDIUM? AI_SPEED_MED : AI_SPEED_HARD;
    }

    void blueAI(Player p, int idx, float spd) {
        // GK stays near goal
        if (p.isGK) {
            float tgx = PX + 40;
            float tgy = clamp(ball.y, PY+PH/2f-GH/2f+10, PY+PH/2f+GH/2f-10);
            moveToward(p, tgx, tgy, spd*1.1f);
            if (dist(p.x,p.y,ball.x,ball.y)<30) autoKick(p, false);
            return;
        }
        // Outfield: midfield/attack positioning
        float targetX, targetY;
        if (ball.x < PX+PW/2f) {
            // Ball on our side — defend
            targetX = p.homeX * 0.8f + ball.x * 0.2f;
            targetY = p.homeY * 0.7f + ball.y * 0.3f;
        } else {
            // Ball on their side — support/attack
            targetX = Math.min(PX+PW-40f, p.homeX + (ball.x-p.homeX)*0.5f);
            targetY = p.homeY*0.5f + ball.y*0.5f;
        }
        moveToward(p, targetX, targetY, spd);
        if (dist(p.x,p.y,ball.x,ball.y)<28) autoKick(p, false);
    }

    void redAI(Player p, int idx, float spd) {
        if (p.isGK) {
            float tgx = PX + PW - 40;
            float tgy = clamp(ball.y, PY+PH/2f-GH/2f+10, PY+PH/2f+GH/2f-10);
            moveToward(p, tgx, tgy, spd*1.1f);
            if (dist(p.x,p.y,ball.x,ball.y)<30) autoKick(p, true);
            return;
        }
        // Red AI chases ball intelligently
        float chaseX, chaseY;
        if (ball.x > PX+PW/2f) {
            // Ball on red side — defend vigorously
            chaseX = ball.x; chaseY = ball.y;
        } else {
            // Attack: move toward blue goal
            if (idx == 4) { // striker
                chaseX = PX + 80; chaseY = PY + PH/2f;
            } else {
                chaseX = p.homeX*0.4f + ball.x*0.6f;
                chaseY = p.homeY*0.5f + ball.y*0.5f;
            }
        }
        moveToward(p, chaseX, chaseY, spd);
        if (dist(p.x,p.y,ball.x,ball.y)<26) autoKick(p, true);
    }

    void moveToward(Player p, float tx, float ty, float spd) {
        float dx=tx-p.x, dy=ty-p.y;
        float d=(float)Math.sqrt(dx*dx+dy*dy);
        if (d<2) { p.state=PlayerState.IDLE; return; }
        float nx=dx/d, ny=dy/d;
        p.x = clamp(p.x+nx*spd, PX+PLAYER_R, PX+PW-PLAYER_R);
        p.y = clamp(p.y+ny*spd, PY+PLAYER_R, PY+PH-PLAYER_R);
        p.facingAngle = (float)Math.atan2(dy,dx);
        p.state = PlayerState.RUN;
    }

    void autoKick(Player p, boolean redTeam) {
        float dx=ball.x-p.x, dy=ball.y-p.y;
        float d=(float)Math.sqrt(dx*dx+dy*dy);
        if (d<3) d=3;
        // Aim toward opponent's goal
        float goalX = redTeam ? PX+10 : PX+PW-10;
        float goalY = PY+PH/2f;
        float shotDx = goalX-ball.x, shotDy = goalY-ball.y;
        float sl=(float)Math.sqrt(shotDx*shotDx+shotDy*shotDy);
        if (sl>0) { shotDx/=sl; shotDy/=sl; }
        float power = 0.5f + rng.nextFloat()*0.4f;
        float spd2  = 7f + power*6f;
        // Add some inaccuracy based on difficulty
        float acc = difficulty==Difficulty.EASY?0.35f:difficulty==Difficulty.MEDIUM?0.18f:0.08f;
        shotDx += (rng.nextFloat()-0.5f)*acc;
        shotDy += (rng.nextFloat()-0.5f)*acc;
        ball.vx = shotDx*spd2; ball.vy = shotDy*spd2;
        p.state = PlayerState.KICK; p.kickAnim=8;
        sound.play("kick");
        if (redTeam) shots[1]++;
    }

    void autoKickAI() {
        // Red players that are very close to ball kick automatically
        for (Player rp : redPlayers) {
            if (dist(rp.x,rp.y,ball.x,ball.y) < PLAYER_R+BALL_R+2 && rng.nextInt(4)==0) {
                autoKick(rp, true);
                break;
            }
        }
        for (int i=0;i<5;i++) {
            if (i==controlledIdx) continue;
            Player bp = bluePlayers[i];
            if (dist(bp.x,bp.y,ball.x,ball.y) < PLAYER_R+BALL_R+2 && rng.nextInt(5)==0) {
                autoKick(bp, false);
                break;
            }
        }
    }

    void tryKick(Player p, float power) {
        float d = dist(p.x,p.y,ball.x,ball.y);
        if (d > PLAYER_R + BALL_R + 18) return; // too far
        // Kick toward where player is facing / toward red goal
        float goalX = PX+PW-10;
        float goalY = PY+PH/2f + (rng.nextFloat()-0.5f)*30;
        float dx=goalX-p.x, dy=goalY-p.y;
        float dl=(float)Math.sqrt(dx*dx+dy*dy);
        if (dl>0){dx/=dl;dy/=dl;}
        float spd2 = 6f + power*12f;
        ball.vx = dx*spd2; ball.vy = dy*spd2;
        p.state=PlayerState.KICK; p.kickAnim=10;
        sound.play("kick");
        shots[0]++;
        spawnKickFx((int)ball.x,(int)ball.y);
    }

    void switchControl() {
        int nearest = nearestBlueIdx();
        controlledIdx = nearest;
        sound.play("switch");
    }

    int nearestBlueIdx() {
        int best=0; float bestD=Float.MAX_VALUE;
        for (int i=0;i<5;i++){
            float d=dist(bluePlayers[i].x,bluePlayers[i].y,ball.x,ball.y);
            if (d<bestD){bestD=d;best=i;}
        }
        return best;
    }

    Player closestToBall(Player[] team){
        Player best=team[0]; float bestD=Float.MAX_VALUE;
        for (Player p:team){float d=dist(p.x,p.y,ball.x,ball.y); if(d<bestD){bestD=d;best=p;}}
        return best;
    }

    void checkGoal() {
        // Left goal (blue goal) — red scores
        if (ball.x <= LGX + GW && ball.y >= GY && ball.y <= GY+GH) {
            scoreRed++;
            lastGoalBlue=false;
            goalScored();
        }
        // Right goal (red goal) — blue scores
        if (ball.x >= RGX - GW && ball.y >= GY && ball.y <= GY+GH) {
            scoreBlue++;
            lastGoalBlue=true;
            shotsOnTarget[0]++;
            goalScored();
        }
    }

    void goalScored() {
        sound.play("goal");
        screen=Screen.GOAL_ANIM;
        goalAnimTimer=0;
        screenShake=20;
        // Confetti
        for (int i=0;i<80;i++) {
            confetti.add(new int[]{
                rng.nextInt(W), rng.nextInt(H/2),
                rng.nextInt(10)+2, rng.nextInt(10)+2,
                rng.nextInt(360),
                rng.nextInt(256), rng.nextInt(256), rng.nextInt(256),
                rng.nextInt(6)-3, 2+rng.nextInt(4)
            });
        }
        addFloat("GOAL!", W/2, H/2-60, lastGoalBlue?TEAM_COLORS[blueTeamIdx][0]:TEAM_COLORS[redTeamIdx][0]);
        // Celebration particles
        for (int i=0;i<40;i++) {
            particles.add(new Particle(W/2, H/2,
                new Color(rng.nextInt(256),rng.nextInt(256),rng.nextInt(256)),
                (float)(rng.nextGaussian()*8),(float)(rng.nextGaussian()*8),
                40+rng.nextInt(30)));
        }
    }

    void updateGoalAnim() {
        goalAnimTimer++;
        // Update confetti
        for (int[] c : confetti) {
            c[0]+=c[8]; c[1]+=c[9];
            c[4]=(c[4]+5)%360;
            if (c[1]>H) { c[0]=rng.nextInt(W); c[1]=-10; }
        }
        if (goalAnimTimer>150){
            confetti.clear();
            kickOff(lastGoalBlue); // team that scored kicks off — actually other team
            screen=Screen.PLAYING;
        }
    }

    void startSecondHalf() {
        half=2; matchTimer=0;
        // Swap sides
        for (Player p:bluePlayers){p.x=W-p.homeX; p.y=p.homeY; p.homeX=W-p.homeX;}
        for (Player p:redPlayers) {p.x=W-p.homeX; p.y=p.homeY; p.homeX=W-p.homeX;}
        ball.x=PX+PW/2; ball.y=PY+PH/2; ball.vx=0; ball.vy=0;
        screen=Screen.PLAYING;
        sound.play("whistle");
    }

    void keepBallInBounds() {
        // Left wall (goal line — not in goal)
        if (ball.x <= PX+BALL_R) {
            if (ball.y < GY || ball.y > GY+GH) { ball.x=PX+BALL_R; ball.vx=Math.abs(ball.vx)*0.6f; }
        }
        // Right wall
        if (ball.x >= PX+PW-BALL_R) {
            if (ball.y < GY || ball.y > GY+GH) { ball.x=PX+PW-BALL_R; ball.vx=-Math.abs(ball.vx)*0.6f; }
        }
        // Top & bottom
        if (ball.y <= PY+BALL_R)    { ball.y=PY+BALL_R;    ball.vy= Math.abs(ball.vy)*0.6f; sound.play("bounce"); }
        if (ball.y >= PY+PH-BALL_R) { ball.y=PY+PH-BALL_R; ball.vy=-Math.abs(ball.vy)*0.6f; sound.play("bounce"); }
    }

    void spawnKickFx(int x, int y) {
        for (int i=0;i<8;i++) particles.add(new Particle(x,y,new Color(255,220,100),
            (float)(rng.nextGaussian()*4),(float)(rng.nextGaussian()*4),10+rng.nextInt(8)));
    }

    float dist(float ax,float ay,float bx,float by){
        float dx=ax-bx,dy=ay-by; return (float)Math.sqrt(dx*dx+dy*dy);
    }

    void drawGoal(Graphics2D g, int x, int y, int gw, int gh, boolean right){
        // Post and crossbar
        g.setColor(new Color(255,255,255,220));
        g.setStroke(new BasicStroke(4f));
        // Backboard
        g.setColor(new Color(255,255,255,30));
        if(!right) g.fillRect(x-gw, y, gw, gh);
        else       g.fillRect(x,    y, gw, gh);
        // Net lines
        g.setColor(new Color(255,255,255,60));
        g.setStroke(new BasicStroke(0.8f));
        for(int gy2=y;gy2<y+gh;gy2+=12){
            if(!right) g.drawLine(x-gw,gy2,x,gy2);
            else       g.drawLine(x,gy2,x+gw,gy2);
        }
        for(int gx2=(right?x:x-gw);gx2<=(right?x+gw:x);gx2+=12)
            g.drawLine(gx2,y,gx2,y+gh);
        // Posts
        g.setColor(Color.WHITE);
        g.setStroke(new BasicStroke(4f));
        g.drawLine(x,y,x,y+gh);
        g.drawLine(x,y,right?x+gw:x-gw,y);
        g.drawLine(x,y+gh,right?x+gw:x-gw,y+gh);
        g.setStroke(new BasicStroke(1f));
    }

    // ─── MATCH SCENE ──────────────────────────────────────────────────
    void drawMatch(Graphics2D g){
        drawStadium(g);
        drawPitch(g);

        // Shadow under players
        for (Player p:redPlayers)  drawPlayerShadow(g,p);
        for (Player p:bluePlayers) drawPlayerShadow(g,p);

        // Ball shadow
        g.setColor(new Color(0,0,0,60));
        g.fillOval((int)(ball.x-BALL_R+3),(int)(ball.y+4),(int)(BALL_R*2),(int)(BALL_R));

        // Players
        for (Player p:redPlayers)  drawPlayer(g,p,false);
        for (Player p:bluePlayers) drawPlayer(g,p,true);

        // Ball
        drawBall(g);

        // Particles
        for (Particle p:particles) p.draw(g);
        for (FloatText f:floaters) f.draw(g);

        drawHUD(g);
    }

    void drawPlayer(Graphics2D g, Player p, boolean isBlue){
        Graphics2D g2=(Graphics2D)g.create();
        g2.translate(p.x, p.y);

        boolean facingRight = Math.cos(p.facingAngle)>0;
        if (!facingRight) g2.scale(-1,1);

        Color shirt  = p.shirtColor;
        Color shorts = p.shortsColor;
        Color skin   = new Color(220,180,130);
        Color skinDk = new Color(180,140,100);

        // Run animation frame
        int frame = (int)(tick/6)%4;
        float legSwing = (p.state==PlayerState.RUN) ? (float)(Math.sin(tick*0.3)*8) : 0;
        float armSwing = (p.state==PlayerState.RUN) ? (float)(Math.sin(tick*0.3)*6) : 0;

        // Kick animation
        float kickLeg = (p.kickAnim>0) ? (float)(-20+p.kickAnim*3) : 0;

        // ── BOOTS ──
        g2.setColor(new Color(30,20,10));
        g2.fillRoundRect(-5,16+2+(int)legSwing,6,5,2,2);  // left boot
        g2.fillRoundRect(-1,16+2-(int)legSwing,6,5,2,2);  // right boot

        // ── SOCKS ──
        g2.setColor(Color.WHITE);
        g2.fillRect(-5,10+(int)legSwing,5,8);
        g2.fillRect(-1,10-(int)legSwing,5,8);

        // ── SHIN GUARDS ──
        g2.setColor(new Color(255,255,240));
        g2.fillRoundRect(-5,8+(int)legSwing,4,6,2,2);
        g2.fillRoundRect(-1,8-(int)legSwing,4,6,2,2);

        // ── SHORTS ──
        g2.setColor(shorts);
        g2.fillRoundRect(-8,4,16,12,4,4);

        // ── LEGS ──
        g2.setColor(skinDk);
        g2.fillRoundRect(-6,4,5,14+(int)legSwing,3,3);  // left leg
        g2.fillRoundRect(1, 4,5,14-(int)legSwing,3,3);  // right leg
        // Kick leg
        if(p.kickAnim>0){
            g2.setColor(shorts);
            g2.fillRoundRect(2,0,(int)kickLeg+4,12,3,3);
        }

        // ── SHIRT ──
        GradientPaint sg=new GradientPaint(-8,-12,shirt.brighter(),8,8,shirt.darker());
        g2.setPaint(sg);
        g2.fillRoundRect(-10,-14,20,20,6,6);

        // Shirt number
        g2.setFont(new Font("Arial",Font.BOLD,8));
        g2.setColor(Color.WHITE);
        FontMetrics fm=g2.getFontMetrics();
        g2.drawString(p.number,-fm.stringWidth(p.number)/2,-3);

        // ── ARMS ──
        g2.setColor(skin);
        // Left arm
        g2.fillRoundRect(-13,-10+(int)armSwing,6,10,3,3);
        g2.setColor(shirt);
        g2.fillRoundRect(-12,-12+(int)armSwing,5,6,2,2);
        // Right arm  
        g2.fillRoundRect(7,-10-(int)armSwing,6,10,3,3);
        g2.setColor(shirt);
        g2.fillRoundRect(7,-12-(int)armSwing,5,6,2,2);

        // ── NECK ──
        g2.setColor(skin);
        g2.fillRect(-3,-17,6,5);

        // ── HEAD ──
        g2.setColor(skin);
        g2.fillOval(-9,-30,18,18);
        g2.setColor(skinDk);
        g2.fillArc(-9,-30,18,9,0,180); // shading

        // Hair (color varies by team)
        g2.setColor(new Color(40,20,10));
        g2.fillRoundRect(-8,-30,16,8,6,6);

        // Face details
        // Eyes
        g2.setColor(Color.WHITE); g2.fillOval(-5,-26,4,3); g2.fillOval(1,-26,4,3);
        g2.setColor(new Color(40,20,10)); g2.fillOval(-4,-25,2,2); g2.fillOval(2,-25,2,2);
        // Mouth
        g2.setColor(new Color(180,80,80));
        if(p.state==PlayerState.KICK) g2.drawOval(-3,-21,6,4); // open mouth
        else g2.drawLine(-2,-20,2,-20);

        // ── GK GLOVES ──
        if(p.isGK){
            g2.setColor(new Color(255,200,0));
            g2.fillRoundRect(-16,-8,6,8,3,3);
            g2.fillRoundRect(10,-8,6,8,3,3);
        }

        // ── CONTROLLED INDICATOR ──
        if(isBlue && p==bluePlayers[controlledIdx]){
            g2.setColor(new Color(255,255,255,160));
            g2.setStroke(new BasicStroke(1.5f));
            g2.drawOval(-PLAYER_R-2,-PLAYER_R-2,(PLAYER_R+2)*2,(PLAYER_R+2)*2);

            int[] ax={0,-5,5}; int[] ay={-34,-40,-40};
            g2.fillPolygon(ax,ay,3);
        }

        // Player name on hover area
        if(isBlue && p==bluePlayers[controlledIdx]){
            g2.setFont(new Font("Arial",Font.BOLD,9));
            g2.setColor(Color.WHITE);
            String n=p.name.split(" ")[0]; // first name only
            g2.drawString(n,-g2.getFontMetrics().stringWidth(n)/2,-36);
        }

        g2.dispose();
    }

    // ─── BALL ─────────────────────────────────────────────────────────
    void drawBall(Graphics2D g){
        int bx=(int)(ball.x-BALL_R), by=(int)(ball.y-BALL_R);
        int bd=(int)(BALL_R*2);

        // Shadow
        g.setColor(new Color(0,0,0,50));
        g.fillOval(bx+3,by+6,bd,bd/2);

        // Base white sphere
        RadialGradientPaint rp=new RadialGradientPaint(
            new Point2D.Float(ball.x-BALL_R*0.3f, ball.y-BALL_R*0.3f), BALL_R,
            new float[]{0f,0.7f,1f},
            new Color[]{Color.WHITE, new Color(220,220,220), new Color(160,160,160)});
        g.setPaint(rp); g.fillOval(bx,by,bd,bd);

        // Spinning pentagon pattern
        Graphics2D g2=(Graphics2D)g.create();
        g2.translate(ball.x,ball.y);
        g2.rotate(ball.spin);
        g2.setColor(new Color(30,30,30));
        // Pentagon patches
        for (int i=0;i<5;i++){
            double a=Math.PI*2*i/5 - Math.PI/2;
            int px2=(int)(Math.cos(a)*BALL_R*0.5f);
            int py2=(int)(Math.sin(a)*BALL_R*0.5f);
            g2.fillOval(px2-3,py2-3,6,6);
        }
        g2.fillOval(-3,-3,6,6); // centre
        g2.dispose();

        // Shine
        g.setColor(new Color(255,255,255,160));
        g.fillOval(bx+2,by+2,bd/3,bd/4);
    }

    @Override public void keyPressed(KeyEvent e){
        int k=e.getKeyCode();
        if(k==KeyEvent.VK_UP)    kUp   =true;
        if(k==KeyEvent.VK_DOWN)  kDown =true;
        if(k==KeyEvent.VK_LEFT)  kLeft =true;
        if(k==KeyEvent.VK_RIGHT) kRight=true;
        if(k==KeyEvent.VK_SPACE) kSpace=true;
        if(k==KeyEvent.VK_TAB)   kTab  =true;

        switch(screen){
            case MENU -> {
                if(k==KeyEvent.VK_UP)    menuSelection=(menuSelection+3)%4;
                if(k==KeyEvent.VK_DOWN)  menuSelection=(menuSelection+1)%4;
                if(k==KeyEvent.VK_ENTER){
                    switch(menuSelection){
                        case 0->{ screen=Screen.TEAM_SELECT; }
                        case 1->{ difficulty=Difficulty.values()[(difficulty.ordinal()+1)%3]; }
                        case 2->screen=Screen.SCORES;
                        case 3->{ /* show controls popup */ }
                    }
                    sound.play("menubeep");
                }
            }
            case TEAM_SELECT -> {
                if(k==KeyEvent.VK_LEFT)  blueTeamIdx=(blueTeamIdx+5)%6;
                if(k==KeyEvent.VK_RIGHT) blueTeamIdx=(blueTeamIdx+1)%6;
                if(k==KeyEvent.VK_UP)    blueTeamIdx=(blueTeamIdx+3)%6;
                if(k==KeyEvent.VK_DOWN)  blueTeamIdx=(blueTeamIdx+3)%6;
                // Auto-assign opponent (different team)
                redTeamIdx=(blueTeamIdx+1)%6;
                if(k==KeyEvent.VK_ENTER){ startMatch(); sound.play("menubeep"); }
                if(k==KeyEvent.VK_ESCAPE) screen=Screen.MENU;
            }
            case PLAYING -> {
                if(k==KeyEvent.VK_P||k==KeyEvent.VK_ESCAPE) screen=Screen.PAUSED;
                if(k==KeyEvent.VK_M) muteSound=!muteSound;
            }
            case PAUSED -> {
                if(k==KeyEvent.VK_P||k==KeyEvent.VK_ESCAPE||k==KeyEvent.VK_ENTER)
                    screen=Screen.PLAYING;
                if(k==KeyEvent.VK_M) muteSound=!muteSound;
            }
            case FULL_TIME -> {
                if(k==KeyEvent.VK_ENTER){ screen=Screen.TEAM_SELECT; }
                if(k==KeyEvent.VK_ESCAPE) screen=Screen.MENU;
            }
            case SCORES -> {
                if(k==KeyEvent.VK_ESCAPE||k==KeyEvent.VK_ENTER) screen=Screen.MENU;
            }
        }
    }

    static float clamp(float v,float lo,float hi){ return Math.max(lo,Math.min(hi,v)); }

    // ══════════════════════════════════════════════════════════════════
    //  INNER CLASS — BALL
    // ══════════════════════════════════════════════════════════════════
    class Ball {
        float x,y,vx,vy,spin,spinRate;
        Ball(float x,float y,float vx,float vy){ this.x=x;this.y=y;this.vx=vx;this.vy=vy; }
        void update(){
            x+=vx; y+=vy;
            vx*=BALL_FRICTION; vy*=BALL_FRICTION;
            // Spin based on velocity
            spinRate=(vx+vy)*0.04f;
            spin+=spinRate;
            // Near-stop
            if(Math.abs(vx)<0.05f) vx=0;
            if(Math.abs(vy)<0.05f) vy=0;
        }
    }

    // ══════════════════════════════════════════════════════════════════
    //  INNER CLASS — SOUND ENGINE (pure Java synthesis)
    // ══════════════════════════════════════════════════════════════════
    class SFX {
        static final int SR=44100;

        void play(String name){
            if(muteSound) return;
            byte[] pcm=gen(name); if(pcm==null) return;
            Thread t=new Thread(()->{
                try{
                    AudioFormat fmt=new AudioFormat(SR,16,1,true,false);
                    DataLine.Info info=new DataLine.Info(SourceDataLine.class,fmt);
                    if(!AudioSystem.isLineSupported(info)) return;
                    try(SourceDataLine line=(SourceDataLine)AudioSystem.getLine(info)){
                        line.open(fmt,2048); line.start();
                        line.write(pcm,0,pcm.length); line.drain();
                    }
                }catch(Exception ignored){}
            });
            t.setDaemon(true); t.start();
        }

        byte[] gen(String name){
            return switch(name){
                // Ball kick — sharp thud
                case "kick"     -> synth(0.14, t->
                    (noise()*0.7+sine(t,120-t*60)*0.5)*env(t,0.005,0.03,0.3,0.06)*0.75);
                // Ball bounce off post/wall
                case "bounce"   -> synth(0.10, t->
                    (sine(t,400-t*200)+noise()*0.2)*env(t,0.005,0.02,0.25,0.05)*0.55);
                // GOAL! — crowd roar + horn
                case "goal"     -> synth(1.80, t->
                    (crowdRoar(t)+hornBlast(t)*0.6)*env(t,0.02,0.15,0.7,0.1)*0.80);
                // Whistle (start / halftime / fulltime)
                case "whistle"  -> synth(0.60, t->
                    (sine(t,2200+sine(t*3,80)*30)+sine(t,2400)*0.3)*env(t,0.01,0.05,0.65,0.1)*0.55);
                // Player switch
                case "switch"   -> synth(0.10, t->
                    sine(t,660)*env(t,0.01,0.03,0.35,0.04)*0.35);
                // Menu beep
                case "menubeep" -> synth(0.12, t->
                    (sine(t,440)+sine(t,880)*0.3)*env(t,0.01,0.03,0.4,0.04)*0.38);
                default         -> null;
            };
        }

        double crowdRoar(double t){
            // Rising noise burst
            return noise()*(0.3+t*0.7)*Math.sin(t*Math.PI);
        }
        double hornBlast(double t){
            if(t<0.3) return sine(t,440+t*200);
            if(t<0.6) return sine(t,660)+sine(t,880)*0.4;
            return sine(t,880)*Math.max(0,1-t);
        }

        static double sine(double t,double f){return Math.sin(2*Math.PI*f*t);}
        static final Random NR=new Random();
        static double noise(){return NR.nextDouble()*2-1;}
        static double env(double t,double a,double d,double s,double r){
            if(t<a) return t/a;
            if(t<a+d) return 1-(1-0.7)*((t-a)/d);
            if(t<1.0-r) return 0.7;
            return 0.7*(1-(t-(1.0-r))/r);
        }
        byte[] synth(double dur,DoubleUnaryOperator fn){
            int n=(int)(SR*dur); byte[] buf=new byte[n*2];
            for(int i=0;i<n;i++){
                double t=(double)i/n;
                int s=(int)(fn.applyAsDouble(t)*28000);
                s=Math.max(-32768,Math.min(32767,s));
                buf[2*i]=(byte)(s&0xFF); buf[2*i+1]=(byte)((s>>8)&0xFF);
            }
            return buf;
        }
    }
